\section{Tổng kết}

\subsection{Kỹ năng và kiến thức thu nhận được}
Qua 8 tuần thực tập thực chiến tại \textbf{Công ty TNHH Amazon Web Services Việt Nam} trong khuôn khổ chương trình \textbf{First Cloud AI Journey 2026}, sinh viên đã tích lũy được nhiều kiến thức và kỹ năng thực tiễn quý giá:
\begin{itemize}
    \item \textbf{Hạ tầng Điện toán đám mây \& DevOps}: Khởi tạo, vận hành và quản lý máy chủ ảo \textbf{Amazon EC2} (Ubuntu 24.04 LTS), cấu hình Elastic IP, Nginx Reverse Proxy, nén Gzip, Swap memory, quản lý tiến trình PM2, CSDL quan hệ \textbf{Amazon RDS MySQL} và tự động hóa CI/CD qua \textbf{GitHub Actions}.
    \item \textbf{Bảo mật \& Phân phối toàn cầu}: Triển khai CDN \textbf{Amazon CloudFront} kết hợp tường lửa \textbf{AWS WAF} (bảo vệ các quy tắc OWASP Top 10 và SQL Injection), tối ưu hóa upload tệp an toàn qua \textbf{Amazon S3 Presigned URLs}.
    \item \textbf{Serverless \& Tự động hóa}: Xây dựng hàm \textbf{AWS Lambda} kết hợp \textbf{Amazon EventBridge Scheduler} tự động bóc tách dữ liệu lịch chiếu phim CGV hàng tuần.
    \item \textbf{Trí tuệ nhân tạo (AI Engine)}: Thiết kế kiến trúc \textbf{Dual-AI Engine} độ sẵn sàng cao kết hợp giữa \textbf{Google Gemini 2.5 Flash} và \textbf{Amazon Bedrock}.
\end{itemize}

\subsection{Cảm nhận cá nhân về doanh nghiệp và chương trình}
\begin{itemize}
    \item \textbf{Về môi trường làm việc}: Văn hóa doanh nghiệp tại AWS Việt Nam cực kỳ cởi mở, chuyên nghiệp, khuyến khích tư duy tự nghiên cứu và giải quyết vấn đề từ gốc rễ (Root Cause Analysis).
    \item \textbf{Về sự hỗ trợ từ Mentor}: Cán bộ hướng dẫn trực tiếp luôn tận tâm chỉ bảo, định hướng kiến trúc đám mây chuẩn hóa và tạo điều kiện tối đa cho sinh viên phát huy năng lực cá nhân.
    \item \textbf{Về giá trị thực tiễn}: Kỳ thực tập giúp thu hẹp khoảng cách giữa lý thuyết đại học và thực tế công nghiệp, giúp sinh viên làm chủ quy trình đưa một sản phẩm hoàn chỉnh lên môi trường sản xuất (\url{https://phuckhanh.id.vn}).
\end{itemize}

\subsection{Phương hướng phát triển trong tương lai}
\begin{itemize}
    \item Nâng cao chuyên sâu về Bảo mật Đám mây (Cloud Security) và áp dụng nguyên tắc quản lý truy cập tối thiểu (IAM Least-Privilege).
    \item Mở rộng hạ tầng dưới dạng mã nguồn (Infrastructure-as-Code - AWS CDK / Terraform) để nâng cao khả năng khôi phục thảm họa tự động.
    \item Định hướng phát triển sự nghiệp trở thành một Kỹ sư Điện toán đám mây / DevOps chuyên nghiệp.
\end{itemize}
